cmake_minimum_required(VERSION 3.22)
project(usbreader LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_subdirectory(erofs)
add_subdirectory(grub)
add_subdirectory(refs)
add_library(usbreader SHARED native_multi.cpp)
target_link_libraries(usbreader PRIVATE grubfs refsfs)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(usbreader PRIVATE Threads::Threads)
endif()
target_link_options(usbreader PRIVATE -Wl,--no-undefined -Wl,--gc-sections -Wl,--exclude-libs,ALL)
if(NOT ANDROID)
    target_include_directories(usbreader PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_compile_options(usbreader PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
